                for (let g = 0; g < numGroups; g++) {

                        groups[g].push(pool[poolIndex]);

                        poolIndex++;

                    } else {

                        break;

                    }

                }

                if (poolIndex >= pool.length) break;

            for (let i = pool.length - 1; i > 0; i--) {

                const j = Math.floor(Math.random() * (i + 1));

            for (let step = 0; step < sizePerGroup; step++) {

                for (let g = 0; g < numGroups; g++) {

                    if (poolIndex < pool.length) {

                        groups[g].push(pool[poolIndex]);

                        poolIndex++;

                    } else {

                        break;

                    }

                }

                if (poolIndex >= pool.length) break;

        // 核心五：富蘭克林決策 & 智慧決定按鈕

        // ==========================================

        let franklinData = {

            question: "",

            pros: [], // Array of { id, text, crossed }

            cons: []  // Array of { id, text, crossed }

        };

        const questionInput = document.getElementById('franklin-question');

        const proInput = document.getElementById('pro-input');

        const conInput = document.getElementById('con-input');

        const proList = document.getElementById('pro-list');

        const conList = document.getElementById('con-list');

        const addProBtn = document.getElementById('add-pro-btn');

        const addConBtn = document.getElementById('add-con-btn');

        const decideFranklinBtn = document.getElementById('decide-franklin-btn');

        const clearFranklinBtn = document.getElementById('clear-franklin-btn');

        function renderFranklin() {

            questionInput.value = franklinData.question;

           

            // 渲染是/優點列表

            proList.innerHTML = '';

            franklinData.pros.forEach((item) => {

                const div = document.createElement('div');

                div.className = `reason-item ${item.crossed ? 'crossed' : ''}`;

                div.innerHTML = `

                    <span>${item.text}</span>

                    <i class="fa-solid fa-circle-xmark delete-reason-btn" data-id="${item.id}" data-type="pro"></i>

                `;

                div.addEventListener('click', (e) => {

                    if (e.target.classList.contains('delete-reason-btn')) return;

                    toggleReasonCrossed(item.id, 'pro');

                });

                proList.appendChild(div);

            });

            // 渲染否/缺點列表

            conList.innerHTML = '';

            franklinData.cons.forEach((item) => {

                const div = document.createElement('div');

                div.className = `reason-item ${item.crossed ? 'crossed' : ''}`;

                div.innerHTML = `

                    <span>${item.text}</span>

                    <i class="fa-solid fa-circle-xmark delete-reason-btn" data-id="${item.id}" data-type="con"></i>

                `;

                div.addEventListener('click', (e) => {

                    if (e.target.classList.contains('delete-reason-btn')) return;

                    toggleReasonCrossed(item.id, 'con');

                });

                conList.appendChild(div);

            });

            // 綁定刪除事件

            document.querySelectorAll('.delete-reason-btn').forEach(btn => {

                btn.addEventListener('click', (e) => {

                    e.stopPropagation();

                    const id = e.target.getAttribute('data-id');

                    const type = e.target.getAttribute('data-type');

                    deleteReason(id, type);

                });

            });

        }

        function addReason(type) {

            if (type === 'pro') {

                const val = proInput.value.trim();

                if (!val) return;

                franklinData.pros.push({ id: crypto.randomUUID(), text: val, crossed: false });

                proInput.value = '';

            } else {

                const val = conInput.value.trim();

                if (!val) return;

                franklinData.cons.push({ id: crypto.randomUUID(), text: val, crossed: false });

                conInput.value = '';

            }

            renderFranklin();

            saveFranklinToFirestore();

        }

        addProBtn.addEventListener('click', () => addReason('pro'));

        addConBtn.addEventListener('click', () => addReason('con'));

        proInput.addEventListener('keypress', (e) => { if (e.key === 'Enter') addReason('pro'); });

        conInput.addEventListener('keypress', (e) => { if (e.key === 'Enter') addReason('con'); });

        function deleteReason(id, type) {

            if (type === 'pro') {

                franklinData.pros = franklinData.pros.filter(item => item.id !== id);

            } else {

                franklinData.cons = franklinData.cons.filter(item => item.id !== id);

            }

            renderFranklin();

            saveFranklinToFirestore();

        }

        function toggleReasonCrossed(id, type) {

            if (type === 'pro') {

                const item = franklinData.pros.find(item => item.id === id);

                if (item) item.crossed = !item.crossed;

            } else {

                const item = franklinData.cons.find(item => item.id === id);

                if (item) item.crossed = !item.crossed;

            }

            renderFranklin();

            saveFranklinToFirestore();

        }

        questionInput.addEventListener('input', () => {

            franklinData.question = questionInput.value;

            saveFranklinToFirestore();

        });

        // 「我決定了！」智慧推薦決策邏輯

        decideFranklinBtn.addEventListener('click', () => {

            const activePros = franklinData.pros.filter(item => !item.crossed).length;

            const activeCons = franklinData.cons.filter(item => !item.crossed).length;

            const questionText = franklinData.question.trim() || "這件大事";

            if (franklinData.pros.length === 0 && franklinData.cons.length === 0) {

                showResult("請先在是與否兩欄中新增一些理由，再讓我幫你分析喔！", "提示");

                return;

            }

            let adviceTitle = "";

            let adviceDetail = `有效優點（是）共 ${activePros} 項，有效缺點（否）共 ${activeCons} 項。`;

            if (activePros > activeCons) {

                adviceTitle = "🟢 建議「去做 / 是」！";

                adviceDetail += `\n支持「是」的理由比較多。如果這是你想追求的目標，就放手去嘗試吧！`;

            } else if (activeCons > activePros) {

                adviceTitle = "🔴 建議「不做 / 否」！";

                adviceDetail += `\n反對的理由或風險點比較多。建議先緩緩或尋找更周全的方案。`;

            } else {

                adviceTitle = "⚖️ 目前旗鼓相當！";

                adviceDetail += `\n優缺點的數量剛好打平。也許可以試著放鬆心情，聽聽內心最直覺的想法，或嘗試用抽籤/轉盤來獲得靈感！`;

            }

            showResult(adviceTitle, `關於「${questionText}」：\n${adviceDetail}`);

        });

        clearFranklinBtn.addEventListener('click', () => {

            franklinData = { question: "", pros: [], cons: [] };

            renderFranklin();

            saveFranklinToFirestore();

        });

                    if (data.options) options = data.options;

            }

            if (!user) return;
